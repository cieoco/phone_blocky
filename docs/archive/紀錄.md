### 🎮 控制邏輯重寫 (src/PS4ControllerHandler.cpp)

#### updateMotorFromStick() 函數完全重寫
```cpp
// 原始複雜版本 (約80行代碼，控制4個馬達映射)
void PS4ControllerHandler::updateMotorFromStick() {
    // 複雜的多馬達控制邏輯
    // 左搖桿控制 leftStickMotor1 和 leftStickMotor2
    // 右搖桿控制 rightStickMotor3 和 rightStickMotor4
    // 各種除錯訊息和狀態管理
}

// 簡化後版本 (約45行代碼，清晰的Y軸控制)
void PS4ControllerHandler::updateMotorFromStick() {
    if (!settingsManager) return;
    
    static unsigned long lastMotorDebugTime = 0;
    static int lastLeftY = 0, lastRightY = 0;
    unsigned long currentTime = millis();
    
    // 取得按鈕映射設定
    PS4SettingsManager::ButtonMapping mapping = settingsManager->getButtonMapping();
    
    // 左搖桿Y軸控制 (僅控制指定的一個馬達)
    if (abs(currentStatus.leftStickY) > 10 && mapping.leftStickMotor > 0) {
        int speed = map(abs(currentStatus.leftStickY), 10, 127, 0, 255);
        char direction = (currentStatus.leftStickY > 0) ? 'F' : 'B';
        
        motorController->controlMotor(mapping.leftStickMotor, direction, speed);
        
        // 除錯訊息
        if (currentTime - lastMotorDebugTime >= 1000 && abs(lastLeftY - currentStatus.leftStickY) > 5) {
            Serial.printf("[PS4] 左搖桿Y軸: %d, 馬達%d, %s, 速度=%d\n", 
                         currentStatus.leftStickY, mapping.leftStickMotor,
                         (direction == 'F') ? "前進" : "後退", speed);
            lastMotorDebugTime = currentTime;
            lastLeftY = currentStatus.leftStickY;
        }
    } else if (mapping.leftStickMotor > 0) {
        // 停止左搖桿控制的馬達
        motorController->controlMotor(mapping.leftStickMotor, 'R', 0);
        
        if (lastLeftY != 0) {
            Serial.println("[PS4] 左搖桿停止");
            lastLeftY = 0;
        }
    }
    
    // 右搖桿Y軸控制 (僅控制指定的一個馬達)
    if (abs(currentStatus.rightStickY) > 10 && mapping.rightStickMotor > 0) {
        int speed = map(abs(currentStatus.rightStickY), 10, 127, 0, 255);
        char direction = (currentStatus.rightStickY > 0) ? 'F' : 'B';
        
        motorController->controlMotor(mapping.rightStickMotor, direction, speed);
        
        if (currentTime - lastMotorDebugTime >= 1000 && abs(lastRightY - currentStatus.rightStickY) > 5) {
            Serial.printf("[PS4] 右搖桿Y軸: %d, 馬達%d, %s, 速度=%d\n", 
                         currentStatus.rightStickY, mapping.rightStickMotor,
                         (direction == 'F') ? "前進" : "後退", speed);
            lastRightY = currentStatus.rightStickY;
        }
    } else if (mapping.rightStickMotor > 0) {
        // 停止右搖桿控制的馬達
        motorController->controlMotor(mapping.rightStickMotor, 'R', 0);
        
        if (lastRightY != 0) {
            Serial.println("[PS4] 右搖桿停止");
            lastRightY = 0;
        }
    }
}
```
